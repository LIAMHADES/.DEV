### **FASE 3: CONEXIONES GPIO (1.5h)**
```
**I2C MPU6050:**
GPIO21 SDA ←── 4.7kΩ ── 3V3
GPIO20 SCL ←── 4.7kΩ ── 3V3
GPIO19 3V3 → MPU6050 VCC
GPIO18 GND → MPU6050 GND

**WS2812B:**
GPIO17 → WS2812B DIN (3V3 OK)
3V3 → WS2812B VCC (470Ω serie)
GND común

**NTC térmica:**
GPIO16 ←── NTC 10kΩ ── 3V3 (divisor)

**SIM7000G control:**
GPIO15 → Gate MOSFET P-CH → 3V8 SIM7000G
GPIO14 → SIM7000G PWRKEY (1s LOW)
GPIO13 → SIM7000G RESET (100ms LOW)
```

### **FASE 4: FIRMWARE BASE (2h)**
```cpp
// FIRMWARE MÍNIMO FUNCIONAL v1.0
#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_NeoPixel.h>

#define WS2812B_PIN 17
#define NTC_PIN 16
#define SIM7000G_PWR 14

Adafruit_NeoPixel led(1, WS2812B_PIN, NEO_GRB + NEO_KHZ800);
Adafruit_MPU6050 mpu;

void setup() {
  Serial.begin(115200);
  
  // Inicializar sistema
  Wire.begin(21, 20);
  pinMode(SIM7000G_PWR, OUTPUT);
  
  // Test LED
  led.begin();
  led.setPixelColor(0, led.Color(0,255,0)); // Verde OK
  led.show();
  
  // Test MPU6050
  if (!mpu.begin()) {
    led.setPixelColor(0, led.Color(255,0,0)); // Rojo FAIL
    Serial.println("MPU6050 fail!");
  }
  
  Serial.println("✓ Tracker v1.0 OK");
}

void loop() {
  // TODO: GPS + MQTT
  delay(1000);
}
```
